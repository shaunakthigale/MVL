a = imread("cameraman.tif");
[counts, ~] = imhist(a);
counts = counts(:)'; 
total = sum(counts);
p = counts / total;
mu_total = sum((0:255) .* p);
Wb = 0;
mu_b = 0;
max_var = 0;
threshold = 0;
for t = 1:256
    Wb = Wb + p(t);
    if Wb == 0
        continue;
    end
       Wf = 1 - Wb;
    if Wf == 0
        break;
    end
    mu_b = mu_b + (t-1) * p(t);
    mu_f = (mu_total - mu_b) / Wf;
        mu_b_norm = mu_b / Wb;
        var_between = Wb * Wf * (mu_b_norm - mu_f)^2;
    
    if var_between > max_var
        max_var = var_between;
        threshold = t-1;
    end
end
fprintf("Otsu Threshold = %d\n", threshold);
binary_img = a > threshold;
subplot(2,1,1)
imshow(a)
title('Original Image')
subplot(2,1,2)
imshow(binary_img)
title(['Binarized Image (Threshold = ', num2str(threshold), ')'])
